{{/* This shortcode is used to display icons in the content. Usage: {{< icons "github" "bootstrap" >}} {{< icons library="bootstrap" name="github" >}} */}} {{- $iconName := .Get 0 | default (.Get "name" | default "") -}} {{- $library := .Get "library" | default (.Get 1 | default "bootstrap") -}} {{- if eq $library "bootstrap" -}} {{- $iconClass := printf "bi bi-%s" $iconName -}} {{- else if eq $library "fontawesome" -}} {{- $iconClass := printf "fa-solid fa-%s" $iconName -}} {{- else if eq $library "heroicons" -}} {{- $iconClass := printf "hi hi-%s" $iconName -}} {{- else if eq $library "feather" -}} {{- $iconClass := printf "fe fe-%s" $iconName -}} {{- else -}} {{ errorf "icon shortcode: unknown library %q. Supported libraries are: bootstrap, fontawesome, heroicons, feather." $library }} {{- end -}} {{- if eq $iconName "" -}} {{ errorf "icon shortcode: missing icon name. Usage: {{< icons \"icon-name\" library=\"bootstrap\" >}}" }} {{- end -}}
